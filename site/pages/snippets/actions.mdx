import { client } from './viem.config'

// Viem batches the concurrent reads into a single multicall.
const [balance, to] = await Promise.all([
  client.token.getBalance({ token: 'usdc' }),
  client.ens.getAddress({ name: 'treasury.wevm.eth' }),
])

const transfer = await client.token.transferSync({
  amount: balance.amount,
  to,
  token: 'usdc',
})
