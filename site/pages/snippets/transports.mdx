```ts twoslash
import { Client, http, loadBalance, rateLimit } from 'viem'
import { mainnet } from 'viem/chains'

const client = Client.create({
  chain: mainnet,
  transport: loadBalance([
    rateLimit(http('https://1.rpc.example'), { requestsPerSecond: 25 }),
    rateLimit(http('https://2.rpc.example'), { requestsPerSecond: 10 }),
  ]),
})
```
