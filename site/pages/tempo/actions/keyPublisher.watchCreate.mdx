# `keyPublisher.watchCreate`

Watches for Key Publisher creation events.

## Usage

:::code-group

```ts twoslash [example.ts]
import { client } from './viem.config'

const unwatch = client.keyPublisher.watchCreate({
  onPublisherCreated: (args, log) => {
    console.log('Publisher ID:', args.publisherId)
    console.log('Owner:', args.owner)
  },
})

// Later, stop watching
unwatch()
```

```ts twoslash [viem.config.ts] filename="viem.config.ts"
// [!include ~/snippets/tempo/viem.config.ts:setup]
```

:::

### Standalone Action

Call the action from `Actions` to pass the Client explicitly.

```ts twoslash
import { Actions } from 'viem/tempo'
import { client } from './viem.config'

const unwatch = Actions.keyPublisher.watchCreate(client, {
  onPublisherCreated: (args, log) => {
    console.log('Publisher ID:', args.publisherId)
    console.log('Owner:', args.owner)
  },
})
```

## Return Value

`() => void`

A function to unsubscribe from the event.

## Parameters

### args (optional)

- **Type:** `object`

```ts
type Args = {
  creator?: Address | Address[] | null
  owner?: Address | Address[] | null
  publisherId?: Hex | Hex[] | null
}
```

Filters events by their indexed fields.

```ts twoslash
import { client } from './viem.config'
// ---cut---
client.keyPublisher.watchCreate({
  args: { owner: '0x742d35Cc6634C0532925a3b844Bc9e7595f0bEbb' }, // [!code focus]
  onPublisherCreated: (args) => console.log(args),
})
```

### fromBlock (optional)

- **Type:** `bigint`

Block to start listening from.

```ts twoslash
import { client } from './viem.config'
// ---cut---
client.keyPublisher.watchCreate({
  fromBlock: 1n, // [!code focus]
  onPublisherCreated: (args) => console.log(args),
})
```

### onError (optional)

- **Type:** `(error: Error) => void`

Callback to invoke when polling for new events fails.

```ts twoslash
import { client } from './viem.config'
// ---cut---
client.keyPublisher.watchCreate({
  onPublisherCreated: (args) => console.log(args),
  onError: (error) => console.error(error), // [!code focus]
})
```

### onPublisherCreated

- **Type:** `(args: Args, log: Log) => void`

```ts
type Args = {
  /** Account that created the publisher. */
  creator: Address
  /** Owner of the publisher. */
  owner: Address
  /** Publisher ID. */
  publisherId: Hex
}
```

Callback to invoke for each event.

```ts twoslash
import { client } from './viem.config'
// ---cut---
client.keyPublisher.watchCreate({
  onPublisherCreated: (args, log) => console.log(args, log), // [!code focus]
})
```

### poll (optional)

- **Type:** `true`

Whether to poll for events instead of subscribing over a WebSocket transport.

```ts twoslash
import { client } from './viem.config'
// ---cut---
client.keyPublisher.watchCreate({
  onPublisherCreated: (args) => console.log(args),
  poll: true, // [!code focus]
})
```

### pollingInterval (optional)

- **Type:** `number`
- **Default:** The Client's `pollingInterval`

Polling frequency, in milliseconds.

```ts twoslash
import { client } from './viem.config'
// ---cut---
client.keyPublisher.watchCreate({
  onPublisherCreated: (args) => console.log(args),
  pollingInterval: 1_000, // [!code focus]
})
```
