# dropTransaction

## Overview

Removes a transaction from the mempool.

## Recipes

### Drop a Transaction


:::code-group

```ts twoslash [example.ts]
import { client } from './viem.config'

await client.txpool.dropTransaction({ // [!code focus]
  hash: '0xe58dceb6b20b03965bb678e27d141e151d7d4efc2334c2d6a49b9fac523f7364', // [!code focus]
}) // [!code focus]
```

```ts twoslash [viem.config.ts] filename="viem.config.ts"
import { Client, http, testActions } from 'viem'
import { anvil } from 'viem/chains'

export const client = Client.create({
  chain: anvil,
  transport: http(),
}).extend(testActions())
```

:::

## `Actions.txpool.dropTransaction`

Removes a transaction from the mempool.

### Usage

Call [`Actions.txpool.dropTransaction`](#actionstxpooldroptransaction) directly by passing the Client as the first argument.

:::code-group

```ts twoslash [example.ts]
import { Actions } from 'viem'
import { client } from './viem.config'

await Actions.txpool.dropTransaction(client, {
  hash: '0xe58dceb6b20b03965bb678e27d141e151d7d4efc2334c2d6a49b9fac523f7364',
})
```

```ts twoslash [viem.config.ts] filename="viem.config.ts"
import { Client, http } from 'viem'
import { anvil } from 'viem/chains'

export const client = Client.create({
  chain: anvil,
  transport: http(),
})
```

:::

### Parameters

#### options.hash

- **Type:** `Hex`

The hash of the transaction to drop.

```ts twoslash
import { Actions, Client, http } from 'viem'
import { mainnet } from 'viem/chains'

const client = Client.create({
  chain: mainnet,
  transport: http(),
})
// ---cut---
await Actions.txpool.dropTransaction(client, {
  hash: '0xe58dceb6b20b03965bb678e27d141e151d7d4efc2334c2d6a49b9fac523f7364', // [!code focus]
  mode: 'hardhat',
})
```

#### options.mode

- **Type:** `'anvil' | 'hardhat' | 'ganache'`
- **Default:** `'anvil'`

The test node implementation to target.

```ts twoslash
import { Actions, Client, http } from 'viem'
import { mainnet } from 'viem/chains'

const client = Client.create({
  chain: mainnet,
  transport: http(),
})
// ---cut---
await Actions.txpool.dropTransaction(client, {
  hash: '0xe58dceb6b20b03965bb678e27d141e151d7d4efc2334c2d6a49b9fac523f7364',
  mode: 'hardhat', // [!code focus]
})
```

### Return Value

`void`

No value is returned.
